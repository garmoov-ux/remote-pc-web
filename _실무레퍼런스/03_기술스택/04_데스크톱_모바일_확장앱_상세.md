## 4. 핵심 실전 구현 코드 3선 (포터블 백엔드 & 오프라인 라이선스)

### 4.1. [실전 코드 1] Electron(`.exe`)이 내장 포터블 Python 서버(`engine.exe`)를 콘솔 창 없이 켜고 끄기 (Sidecar 패턴)
윈도우에서 `PyInstaller -F`로 빌드된 파이썬 `.exe`는 실행 시 내부적으로 자식 프로세스를 한 번 더 생성하므로, 앱을 종료할 때 일반 `proc.kill()` 대신 **`taskkill /pid <PID> /t /f`(하위 프로세스 트리 전체 강제 종료)**를 호출해야 백그라운드 좀비 프로세스가 남지 않습니다.

```typescript
// electron/main.ts
import { app, BrowserWindow } from 'electron';
import { spawn, execSync, ChildProcess } from 'child_process';
import path from 'path';

let pyProc: ChildProcess | null = null;
let mainWindow: BrowserWindow | null = null;

function startPythonBackend() {
  // 개발 중에는 로컬 빌드 경로, 배포 후(packaged)에는 resources/bin/ 경로 참조
  const exePath = app.isPackaged
    ? path.join(process.resourcesPath, 'bin', 'app_engine.exe')
    : path.join(__dirname, '..', 'backend', 'dist', 'app_engine.exe');

  pyProc = spawn(exePath, ['--port', '18000'], {
    windowsHide: true, // 검은 콘솔 창 완벽 숨김
    stdio: 'ignore',
  });
}

function stopPythonBackend() {
  if (pyProc && pyProc.pid) {
    try {
      // 윈도우에서 PyInstaller 자식 프로세스 트리까지 남김없이 종료 (/t: 트리 전체, /f: 강제)
      execSync(`taskkill /pid ${pyProc.pid} /t /f`, { stdio: 'ignore' });
    } catch {
      pyProc.kill();
    }
    pyProc = null;
  }
}

app.whenReady().then(() => {
  startPythonBackend();

  mainWindow = new BrowserWindow({
    width: 1440,

    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,

    },
  });

  mainWindow.loadURL('http://127.0.0.1:18000');
});

app.on('will-quit', () => {
  stopPythonBackend();
});
```

### 4.2. [실전 코드 2] Node.js 없이 순수 파이썬만으로 HTML/Tailwind 데스크톱 `.exe` 띄우기 (`Pywebview` 포터블 패턴)
Electron 없이 파이썬 하나만으로 모던 웹 화면을 윈도우 네이티브 창으로 띄우고, 자바스크립트에서 파이썬 함수를 직접 호출하는 포터블 코드입니다.

```python
# main.py (uv add pywebview 후 uv run pyinstaller -F -w main.py 로 단일 .exe 빌드)
import webview
import sqlite3
from pathlib import Path

class PortableBackendApi:
    """HTML/JS 화면에서 window.pywebview.api.<함수명>() 으로 직접 호출되는 파이썬 백엔드 클래스"""
    def __init__(self):
        self.db_path = Path("./data/portable.db")
        self.db_path.parent.mkdir(exist_ok=True)
        self._init_db()

    def _init_db(self):
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("PRAGMA journal_mode=WAL;")
            conn.execute("CREATE TABLE IF NOT EXISTS logs (id INTEGER PRIMARY KEY, msg TEXT);")

    def save_and_get_count(self, message: str) -> dict:
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("INSERT INTO logs (msg) VALUES (?)", (message,))
            count = conn.execute("SELECT COUNT(*) FROM logs").fetchone()[0]
        return {"ok": True, "totalCount": count}

if __name__ == "__main__":
    api = PortableBackendApi()
    window = webview.create_window(
        title="포터블 통합 관리 도구",
        url="templates/index.html",  # 또는 내장 FastAPI 주소 "http://127.0.0.1:18000"
        js_api=api,
        width=1280,
        height=820,
    )
    webview.start()
```

### 4.3. [실전 코드 3] 인터넷이 안 되는 폐쇄망 납품용 `.exe` 하드웨어 고유번호(UUID) 라이선스 잠금
외부 인터넷이 차단된 기업이나 기관 PC에 포터블 `.exe`를 납품할 때, 허가받지 않은 다른 PC로 파일을 무단 복사해서 실행하는 것을 막는 오프라인 라이선스 검증 코드입니다.

```python
# src/utils/offline_license.py
import hashlib
import subprocess

SECRET_SALT = "APP-2026-MASTER-SECRET-KEY"

def get_windows_machine_uuid() -> str:
    """윈도우 메인보드 고유 UUID 추출 (포맷 전까지 변하지 않음)"""
    cmd = "powershell -NoProfile -Command \"(Get-CimInstance -Class Win32_ComputerSystemProduct).UUID\""
    out = subprocess.check_output(cmd, shell=True, text=True).strip()
    return out

def generate_license_key_for_pc(machine_uuid: str, expire_ymd: str = "20271231") -> str:
    """[개발자용] 고객 PC의 UUID와 만료일(YYYYMMDD)을 넣어 시리얼 키 생성"""
    raw = f"{machine_uuid}:{expire_ymd}:{SECRET_SALT}".encode("utf-8")
    sig = hashlib.sha256(raw).hexdigest()[:16].upper()
    return f"{expire_ymd}-{sig[:4]}-{sig[4:8]}-{sig[8:12]}-{sig[12:16]}"

def verify_offline_license(license_key: str, current_ymd: str) -> bool:
    """[납품 프로그램 내장용] 현재 PC의 UUID 및 만료일과 시리얼 키가 일치하는지 오프라인 검증"""
    try:
        expire_ymd = license_key.split("-")[0]
        if current_ymd > expire_ymd:
            return False  # 사용 기간 만료
        expected_key = generate_license_key_for_pc(get_windows_machine_uuid(), expire_ymd)
        return license_key.strip().upper() == expected_key
    except Exception:
        return False
```
