## 1. `openpyxl` 실무 표준 엑셀(`.xlsx`) 생성 함수

행 높이 `25`, 글꼴 `맑은 고딕 10pt`, 1행 제목 굵게(Bold) 및 배경색(`#D9E1F2`), 모든 테두리(`#BFBFBF`), 첫 행 틀 고정(`A2`), 자동 필터, 한글 글자 폭(1.8배)을 반영한 열 너비 자동 조절, 그리고 동일 그룹 간 배경색 교차(줄무늬)까지 한 번에 처리하는 표준 템플릿입니다.

```python
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter

def create_styled_excel(
    file_path: str,
    sheet_title: str,
    headers: list[str],
    rows: list[list],
    left_align_cols: set[str] | None = None,
    right_align_cols: set[str] | None = None,
    group_col_idx: int | None = None,
):
    """
    실무 표준 서식(행높이 25, 맑은 고딕 10pt, 제목 굵게, 모든 테두리, 한글 열너비 자동맞춤)으로 엑셀 저장
    """
    left_align_cols = left_align_cols or set()
    right_align_cols = right_align_cols or set()

    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = sheet_title

    # 1. 데이터 입력
    ws.append(headers)
    for row in rows:
        ws.append(row)

    max_row = ws.max_row
    max_col = ws.max_column

    # 2. 공통 스타일 정의
    font_header = Font(name="맑은 고딕", size=10, bold=True, color="000000")
    font_bold = Font(name="맑은 고딕", size=10, bold=True, color="000000")
    font_data = Font(name="맑은 고딕", size=10, bold=False, color="000000")

    thin_side = Side(border_style="thin", color="BFBFBF")
    border_all = Border(left=thin_side, right=thin_side, top=thin_side, bottom=thin_side)

    fill_header = PatternFill(start_color="D9E1F2", end_color="D9E1F2", fill_type="solid")
    fill_white = PatternFill(start_color="FFFFFF", end_color="FFFFFF", fill_type="solid")
    fill_stripe = PatternFill(start_color="F2F6FC", end_color="F2F6FC", fill_type="solid")

    align_center = Alignment(horizontal="center", vertical="center", wrap_text=False)
    align_left = Alignment(horizontal="left", vertical="center", wrap_text=False)
    align_right = Alignment(horizontal="right", vertical="center", wrap_text=False)

    # 3. 첫 행 틀 고정 및 자동 필터 적용
    ws.freeze_panes = "A2"
    ws.auto_filter.ref = f"A1:{get_column_letter(max_col)}{max_row}"

    # 4. 1행(제목) 서식 적용 (행높이 25)
    ws.row_dimensions[1].height = 25
    for col_idx in range(1, max_col + 1):
        cell = ws.cell(row=1, column=col_idx)
        cell.font = font_header
        cell.fill = fill_header
        cell.border = border_all
        cell.alignment = align_center

    # 5. 데이터 행 서식 적용 (행높이 25)
    prev_group_val = None
    use_stripe = False

    for row_idx in range(2, max_row + 1):
        ws.row_dimensions[row_idx].height = 25

        if group_col_idx is not None:
            curr_group_val = ws.cell(row=row_idx, column=group_col_idx).value
            if curr_group_val != prev_group_val:
                use_stripe = not use_stripe
                prev_group_val = curr_group_val
        row_fill = fill_stripe if use_stripe else fill_white

        for col_idx in range(1, max_col + 1):
            cell = ws.cell(row=row_idx, column=col_idx)
            cell.font = font_bold if col_idx == group_col_idx else font_data
            cell.fill = row_fill
            cell.border = border_all

            col_name = headers[col_idx - 1]
            if col_name in left_align_cols:
                cell.alignment = align_left
            elif col_name in right_align_cols:
                cell.alignment = align_right
            else:
                cell.alignment = align_center

    # 6. 한글 글자 폭(1.8배)을 반영한 열 너비 자동 맞춤
    for col_idx in range(1, max_col + 1):
        col_letter = get_column_letter(col_idx)
        max_len = 0
        for row_idx in range(1, max_row + 1):
            val = str(ws.cell(row=row_idx, column=col_idx).value or "")
            w = sum(1.8 if ord(ch) > 127 else 1.0 for ch in val)
            if w > max_len:
                max_len = w
        ws.column_dimensions[col_letter].width = min(max(max_len + 3.5, 9), 65)

    wb.save(file_path)
```

## 2. CSV 한글 인코딩(`utf-8-sig` / `cp949`) 안전 입출력 함수

```python
import csv

def read_csv_safe(file_path: str) -> list[dict]:
    """UTF-8(BOM 포함)과 CP949(EUC-KR)를 자동으로 판별하여 안전하게 CSV 읽기"""
    for enc in ("utf-8-sig", "cp949", "euc-kr"):
        try:
            with open(file_path, mode="r", encoding=enc, newline="") as f:
                return list(csv.DictReader(f))
        except UnicodeDecodeError:
            continue
    raise ValueError(f"지원하지 않는 인코딩입니다: {file_path}")

def write_csv_for_excel(file_path: str, fieldnames: list[str], rows: list[dict]):
    """윈도우 엑셀에서 바로 열어도 한글이 깨지지 않도록 utf-8-sig로 저장"""
    with open(file_path, mode="w", encoding="utf-8-sig", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)
```

## 3. 위도 및 경도 실제 거리 계산(Haversine) 및 반경 `N`미터 그룹화

```python
import math
from collections import defaultdict

def haversine_meters(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    """두 위경도 좌표(도 단위) 사이의 실제 지표면 거리(미터, m) 반환"""
    R = 6371000.0
    phi1, phi2 = math.radians(lat1), math.radians(lat2)
    dphi = math.radians(lat2 - lat1)
    dlambda = math.radians(lon2 - lon1)
    a = (math.sin(dphi / 2.0) ** 2
         + math.cos(phi1) * math.cos(phi2) * math.sin(dlambda / 2.0) ** 2)
    return 2.0 * R * math.atan2(math.sqrt(a), math.sqrt(1.0 - a))

def group_by_radius(items: list[dict], radius_m: float = 400.0, strict_all_pairs: bool = False):
    """
    items: [{'id': 'STORE-01', 'lat': 37.56, 'lon': 126.97, ...}, ...]
    strict_all_pairs=False: 인접 연결(A-B <= 400m, B-C <= 400m 이면 동일 그룹)
    strict_all_pairs=True : 완전 연결(그룹 내 모든 지점 쌍의 거리가 radius_m 이하일 때만 동일 그룹)
    """
    n = len(items)
    if not strict_all_pairs:
        adj = defaultdict(list)
        for i in range(n):
            for j in range(i + 1, n):
                d = haversine_meters(items[i]["lat"], items[i]["lon"], items[j]["lat"], items[j]["lon"])
                if d <= radius_m:
                    adj[i].append((j, d))
                    adj[j].append((i, d))

        visited = [False] * n
        groups, singletons = [], []
        for i in range(n):
            if not visited[i]:
                if i in adj:
                    comp, q = [], [i]
                    visited[i] = True
                    while q:
                        curr = q.pop(0)
                        comp.append(items[curr])
                        for nxt, _ in adj[curr]:
                            if not visited[nxt]:
                                visited[nxt] = True
                                q.append(nxt)
                    groups.append(comp)
                else:
                    visited[i] = True
                    singletons.append(items[i])
        return groups, singletons
    else:
        clusters = [[i] for i in range(n)]
        while True:
            best_d = float("inf")
            best_pair = None
            for i in range(len(clusters)):
                for j in range(i + 1, len(clusters)):
                    max_d = max(
                        haversine_meters(items[u]["lat"], items[u]["lon"], items[v]["lat"], items[v]["lon"])
                        for u in clusters[i] for v in clusters[j]
                    )
                    if max_d <= radius_m and max_d < best_d:
                        best_d = max_d
                        best_pair = (i, j)
            if best_pair is None:
                break
            i, j = best_pair
            clusters[i].extend(clusters[j])
            clusters.pop(j)

        groups = [[items[u] for u in c] for c in clusters if len(c) >= 2]
        singletons = [items[c[0]] for c in clusters if len(c) == 1]
        return groups, singletons
```

## 4. 장비 통신용 바이너리 패킷(`struct`) 조립, 언패킹 및 헥사 덤프

### 4.1. `struct` 주요 포맷 문자 정의
- **`>` (Big-Endian)**: 네트워크 표준 바이트 순서 (상위 바이트 먼저 전송)
- **`<` (Little-Endian)**: x86/Windows 메모리 바이트 순서 (하위 바이트 먼저 전송)
- **`B` / `b` (1 Byte)**: 부호 없는 정수(`0~255`) / 부호 있는 정수(`-128~127`)
- **`H` / `h` (2 Bytes)**: 부호 없는 2바이트 정수(`0~65535`) / 부호 있는 2바이트 정수
- **`I` / `i` (4 Bytes)**: 부호 없는 4바이트 정수 / 부호 있는 4바이트 정수
- **`Q` / `q` (8 Bytes)**: 부호 없는 8바이트 정수 / 부호 있는 8바이트 정수
- **`f` / `d` (4 / 8 Bytes)**: 단정밀도 실수(Float) / 배정밀도 실수(Double)
- **`Ns` (N Bytes)**: 고정 길이 바이트 문자열 (예: `5s` = 5바이트)

### 4.2. 고정 길이 바이트 패딩/언패킹 및 헥사 덤프 함수
```python
import struct

def pack_fixed_str(text: str, length: int, encoding: str = "cp949") -> bytes:
    """문자열을 지정된 바이트 길이(length)로 인코딩하고 남는 공간은 NULL(\x00)로 채움"""
    raw = text.encode(encoding, errors="ignore")[:length]
    return raw.ljust(length, b"\x00")

def unpack_fixed_str(raw_bytes: bytes, encoding: str = "cp949") -> str:
    """고정 길이 바이트에서 뒤쪽 NULL(\x00)과 공백을 제거하고 문자열로 복원"""
    return raw_bytes.split(b"\x00", 1)[0].decode(encoding, errors="replace").strip()

def hex_dump(data: bytes) -> str:
    """디버깅 로그용: 바이트 데이터를 16진수(02 A1 FF ...) 문자열로 변환"""
    return " ".join(f"{b:02X}" for b in data)

# 사용 예시: [STX(1B) | 노드ID(5B) | 명령코드(1B) | 바디길이(4B, Big-Endian)] 총 11바이트 헤더
HEADER_FMT = ">B5sBI"
packet_header = struct.pack(HEADER_FMT, 0x02, pack_fixed_str("SRV01", 5), 0x11, 256)
print("송신 헤더 Hex:", hex_dump(packet_header))

stx, node_bytes, cmd_code, body_len = struct.unpack(HEADER_FMT, packet_header)
node_id = unpack_fixed_str(node_bytes)
```

## 5. 실무 필수 정규표현식 (`re`) 6선

```python
import re

# 1. 이메일 주소 추출 및 포맷 검증
EMAIL_RE = re.compile(r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}\b")

# 2. 시스템/노드 식별자 추출 (예: SRV01, A1024 등 영문 1~3자리 + 숫자 2~4자리)
NODE_ID_RE = re.compile(r"\b[A-Z]{1,3}\d{2,4}\b")

# 3. 표준 로그/백업 파일명(유형_노드ID_시각_세션.확장자) 구조 검증 및 그룹 추출
LOG_FILENAME_RE = re.compile(
    r"^(?P<type>[A-Z0-9]{2,4})_(?P<node>[A-Z0-9]{5})_(?P<time>\d{14,17})_(?P<seq>[A-Z0-9]{2})\.(?:LOG|JSON|ZIP|PNG)$",
    re.IGNORECASE,
)

# 4. IPv4 주소 추출 (예: 192.168.0.100)
IPV4_RE = re.compile(r"\b(?:\d{1,3}\.){3}\d{1,3}\b")

# 5. 괄호 안의 부가 설명 텍스트만 추출 (예: "서울센터(메인게이트웨이)" -> "메인게이트웨이")
PAREN_CONTENT_RE = re.compile(r"\(([^)]+)\)")

# 6. 문자열에서 숫자(정수/소수)만 전부 추출
NUMBERS_RE = re.compile(r"-?\d+(?:\.\d+)?")
```
